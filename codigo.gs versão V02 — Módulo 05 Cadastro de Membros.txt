function tzcM05_autorizarV02_(token){
 const sess=tzcM02_sessaoV03_(String(token||''));
 if(!sess||!sess.email)return {ok:false,mensagem:'Entre novamente para continuar.'};
 const u=tzcM02_buscarUsuarioV03_(sess.email);
 if(!u||String(u.status||'').toLowerCase()!=='ativo')return {ok:false,mensagem:'Seu acesso não está ativo.'};
 const teste=typeof TZC_M02_CFG_V03!=='undefined'&&TZC_M02_CFG_V03.MODO_TESTE===true;
 if(!teste&&!['dirigente','administrador'].includes(String(u.perfil||'').toLowerCase()))
  return {ok:false,mensagem:'Somente a gestão pode acessar este cadastro.'};

}

function tzcM05_textoV02_(v,max){return String(v==null?'':v).trim().slice(0,max||2000)}
function tzcM05_emailV02_(v){const x=String(v||'').trim().toLowerCase();return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(x)?x:''}
function tzcM05_dataBRV02_(v){
 if(v instanceof Date&&!isNaN(v.getTime()))return Utilities.formatDate(v,TZC_M05_CFG_V02.FUSO,'dd/MM/yyyy');
 const s=String(v==null?'':v).trim();let a=s.match(/^(\d{4})[-/](\d{1,2})[-/](\d{1,2})/);
 if(a)return ('0'+a[3]).slice(-2)+'/'+('0'+a[2]).slice(-2)+'/'+a[1];
 a=s.match(/^(\d{1,2})[-/](\d{1,2})[-/](\d{4})/);
 return a?('0'+a[1]).slice(-2)+'/'+('0'+a[2]).slice(-2)+'/'+a[3]:s;
}
function tzcM05_dadosLinhaV02_(row,map){const o={};Object.keys(map).forEach(k=>o[k]=row[map[k]]);return o}

function tzcM05_atualizarUsuarioV02_(ss,member,id){
 const sh=ss.getSheetByName(TZC_M05_CFG_V02.ABA_USUARIOS);if(!sh)return;
 const map=tzcM05_mapaV02_(sh);if(map.ID_Filho==null||map.Email==null)return;
 const rows=sh.getLastRow()>1?sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getValues():[];
 for(let i=0;i<rows.length;i++){
  const r=rows[i],sameId=String(r[map.ID_Filho]||'')===String(id),memberEmail=tzcM05_emailV02_(member.Email),sameEmail=!!memberEmail&&tzcM05_emailV02_(r[map.Email])===memberEmail;
  if(!sameId&&!sameEmail)continue;
  if(map.ID_Filho!=null)r[map.ID_Filho]=id;
  if(map.Email!=null&&member.Email)r[map.Email]=member.Email;
  if(map.Nome!=null)r[map.Nome]=member.Nome_Completo;
  if(map.Perfil!=null)r[map.Perfil]=member.Perfil||'Membro';
  if(map.Status!=null)r[map.Status]=member.Status_Ativo||'Ativo';
  sh.getRange(i+2,1,1,r.length).setValues([r]);
 }
}
/** Retorna todos os campos ao editor autenticado; a tabela resumida usa só colunas de gestão. */
function TZC_M05_listarMembrosV02(token){
 const a=tzcM05_autorizarV02_(token);if(!a.ok)return a;
 try{
  const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID),sh=ss.getSheetByName(TZC_M05_CFG_V02.ABA_MEMBROS);
  if(!sh||sh.getLastRow()<2)return {ok:true,membros:[],opcoesFuncao:TZC_M05_CFG_V02.FUNCOES_BASE};
  const map=tzcM05_mapaV02_(sh),rows=sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getDisplayValues(),fun=[];
  const membros=rows.map(r=>tzcM05_dadosLinhaV02_(r,map)).filter(x=>x.ID_Filho&&x.Nome_Completo).map(x=>{
   const o={};TZC_M05_CAMPOS_V02.forEach(k=>o[k]=x[k]==null?'':String(x[k]));
   o.ID_Filho=String(x.ID_Filho);o.Data_Nascimento=tzcM05_dataBRV02_(o.Data_Nascimento);o.Data_Entrada=tzcM05_dataBRV02_(o.Data_Entrada);
   if(o.Cargo_Funcao&&fun.indexOf(o.Cargo_Funcao)<0)fun.push(o.Cargo_Funcao);return o;
  }).sort((x,y)=>x.Nome_Completo.localeCompare(y.Nome_Completo,'pt-BR',{sensitivity:'base'}));
  return {ok:true,membros:membros,opcoesFuncao:Array.from(new Set(TZC_M05_CFG_V02.FUNCOES_BASE.concat(fun))).sort((x,y)=>x.localeCompare(y,'pt-BR'))};
 }catch(e){console.error('M05 listar V02: '+e.message);return {ok:false,mensagem:'Não foi possível carregar os membros.'}}
}
/** Cria ou atualiza sem trocar ID; grava os campos da ficha e só sincroniza contas já existentes. */
function TZC_M05_salvarMembroV02(token,json){
 const a=tzcM05_autorizarV02_(token);if(!a.ok)return a;
 let p;try{p=JSON.parse(String(json||''))}catch(e){return {ok:false,mensagem:'Dados do cadastro inválidos.'}}
 const id=tzcM05_textoV02_(p.id,100),pendingId=tzcM05_textoV02_(p.pendingId,100),fields=p.fields||{};
 const nome=tzcM05_textoV02_(fields.Nome_Completo,150),email=tzcM05_emailV02_(fields.Email);
 if(!nome)return {ok:false,mensagem:'Informe o nome completo.'};
 if(fields.Email&&!email)return {ok:false,mensagem:'Confira o formato do e-mail.'};
 const lock=LockService.getScriptLock();
 try{
  lock.waitLock(8000);
  const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID),sh=ss.getSheetByName(TZC_M05_CFG_V02.ABA_MEMBROS);
  if(!sh)throw Error('Cadastro_Filhos não encontrada.');
  const map=tzcM05_mapaV02_(sh),rows=sh.getLastRow()>1?sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getValues():[];
  let rowNo=0,before=null;
  rows.forEach((r,i)=>{const o=tzcM05_dadosLinhaV02_(r,map);if(id&&String(o.ID_Filho)===id){rowNo=i+2;before=o}});
  if(id&&!rowNo)return {ok:false,mensagem:'Este cadastro não existe mais. Atualize a lista.'};
  rows.forEach(r=>{const o=tzcM05_dadosLinhaV02_(r,map);if(String(o.ID_Filho||'')!==id&&email&&tzcM05_emailV02_(o.Email)===email)throw Error('Este e-mail já está cadastrado para outra pessoa.')});
  const wasExisting=!!rowNo;
  const memberId=id||('FILHO_'+Utilities.getUuid().replace(/-/g,'').slice(0,16));
  const current=rowNo?sh.getRange(rowNo,1,1,sh.getLastColumn()).getValues()[0]:new Array(sh.getLastColumn()).fill('');
  const after={};
  TZC_M05_CAMPOS_V02.forEach(k=>{
   if(map[k]==null)return;
   let v=fields[k]==null?'':fields[k];
   if(k==='Email')v=email;
   else if(k==='Nome_Completo')v=nome;
   else if(k==='Data_Entrada'||k==='Data_Nascimento')v=tzcM05_dataBRV02_(v);
   else if(k==='Pode_Cadastrar_Mensagem_Dia'||k==='Pode_Aprovar_Mensagem_Dia')v=String(v).toLowerCase()==='sim'?'Sim':'Não';
   else v=tzcM05_textoV02_(v,k==='Alergias_Problemas_Saude'||k==='Observacoes'?2000:500);
   current[map[k]]=v;after[k]=v;
  });
  if(map.ID_Filho==null)throw Error('Falta a coluna ID_Filho.');
  current[map.ID_Filho]=memberId;
  if(!rowNo&&map.Status_Ativo!=null&&!after.Status_Ativo)current[map.Status_Ativo]='Ativo';
  if(!rowNo&&map.Perfil!=null&&!after.Perfil)current[map.Perfil]='Membro';
  if(rowNo)sh.getRange(rowNo,1,1,current.length).setValues([current]);
  else{sh.appendRow(current);rowNo=sh.getLastRow()}
  const saved=Object.assign({},after,{ID_Filho:memberId});
  tzcM05_atualizarUsuarioV02_(ss,saved,memberId);
  if(pendingId){
   const ps=ss.getSheetByName(TZC_M05_CFG_V02.ABA_PENDENTES),pm=tzcM05_mapaV02_(ps);
   const pr=ps.getLastRow()>1?ps.getRange(2,1,ps.getLastRow()-1,ps.getLastColumn()).getValues():[];
   for(let i=0;i<pr.length;i++)if(String(pr[i][pm.ID_Solicitacao]||'')===pendingId){
    if(pm.Status!=null)pr[i][pm.Status]='Aprovado';
    if(pm.Decidido_Em!=null)pr[i][pm.Decidido_Em]=new Date();
    if(pm.Decidido_Por!=null)pr[i][pm.Decidido_Por]=a.usuario.email||'';
    if(pm.ID_Filho!=null)pr[i][pm.ID_Filho]=memberId;
    ps.getRange(i+2,1,1,pr[i].length).setValues([pr[i]]);break;
   }
  }
  tzcM05_auditarV02_(ss,a.usuario,wasExisting?'EDITAR':'INCLUIR',memberId,before,saved);
  return {ok:true,mensagem:pendingId?'Ficha revisada e cadastrada.':id?'Cadastro atualizado.':'Membro cadastrado.',id:memberId};
 }catch(e){console.error('M05 salvar V02: '+e.message);return {ok:false,mensagem:e.message==='Este e-mail já está cadastrado para outra pessoa.'?e.message:'Não foi possível salvar o cadastro agora.'}}
 finally{try{lock.releaseLock()}catch(ignore){}}
}
