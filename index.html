** * Varre todas as abas da planilha procurando colunas que terminam com "_URL". * Para cada coluna "_URL", localiza a coluna correspondente de origem (sem o sufixo _URL), * busca o arquivo no Google Drive, concede permissão pública e insere o link gerado. */ function processarTodasTabelasURL() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var abas = ss.getSheets(); abas.forEach(function(aba) { var nomeAba = aba.getName(); var dados = aba.getDataRange().getValues(); if (dados.length <= 1) return; // Aba vazia ou só com cabeçalho var cabecalho = dados[0]; var mapeamentoColunas = []; // 1. Mapear todas as colunas que terminam com "_URL" for (var colIdx = 0; colIdx < cabecalho.length; colIdx++) { var nomeColuna = cabecalho[colIdx].toString().trim(); if (nomeColuna.toUpperCase().endsWith("_URL")) { // Exemplo: De "Imagem_URL" obtém "Imagem" / De "Video_URL" obtém "Video" var nomeColunaOrigem = nomeColuna.replace(/_URL$/i, ""); var colOrigemIdx = cabecalho.indexOf(nomeColunaOrigem); // Se encontrou a coluna de origem correspondente sem o sufixo _URL if (colOrigemIdx !== -1) { mapeamentoColunas.push({ origemIdx: colOrigemIdx, destinoIdx: colIdx, nomeOrigem: nomeColunaOrigem, nomeDestino: nomeColuna }); } } } if (mapeamentoColunas.length === 0) return; // Nenhuma coluna _URL válida nesta aba Logger.log("Processando aba: " + nomeAba + " | Colunas com _URL encontradas: " + mapeamentoColunas.length); // 2. Processar linha por linha da aba for (var i = 1; i < dados.length; i++) { mapeamentoColunas.forEach(function(mapa) { var caminhoOuNome = dados[i][mapa.origemIdx]; if (caminhoOuNome && caminhoOuNome.toString().trim() !== "") { // Extrai o nome do arquivo se for um caminho completo (ex: "Mural_Imagens/foto.jpg" -> "foto.jpg") var nomeArquivo = caminhoOuNome.toString().split("/").pop().trim(); var arquivos = DriveApp.getFilesByName(nomeArquivo); if (arquivos.hasNext()) { var arq = arquivos.next(); // Define permissão pública de visualização com o link arq.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var fileId = arq.getId(); var mimeType = arq.getMimeType(); var urlFinal = ""; // Se for imagem, usa o link direto do Googleusercontent (lh3) if (mimeType.indexOf("image") !== -1 || /\.(jpg|jpeg|png|gif|webp)$/i.test(nomeArquivo)) { urlFinal = "https://lh3.googleusercontent.com/d/" + fileId; } else { // Se for vídeo ou áudio, usa o link de preview/player do Google Drive urlFinal = "https://drive.google.com/file/d/" + fileId + "/preview"; } // Grava a URL gerada na coluna de destino (_URL) aba.getRange(i + 1, mapa.destinoIdx + 1).setValue(urlFinal); } } }); } }); Logger.log("Processamento de colunas _URL concluído!"); } function redimensionarImagensAppSheet() { // COLOQUE AQUI O ID DA PASTA DO GOOGLE DRIVE ONDE O APPSHEET SALVA AS FOTOS var ID_DA_PASTA = "1QwjZSVePFYIRi7sa2SYPsL5cyQsPPira"; var pasta = DriveApp.getFolderById(ID_DA_PASTA); var arquivos = pasta.getFiles(); while (arquivos.hasNext()) { var arquivo = arquivos.next(); var tipo = arquivo.getMimeType(); // Processa apenas imagens JPG ou PNG if (tipo === "image/jpeg" || tipo === "image/png") { // Verifica se a imagem já foi processada para não refazer if (!arquivo.getName().startsWith("OK_")) { var blob = arquivo.getBlob(); // Define a largura máxima desejada (ex: 1200px para manter alta qualidade) var larguraDesejada = 1200; // Redimensiona usando a API de imagens do Google var imagemRedimensionada = ImagesService.resize(blob, larguraDesejada, 800); // Atualiza o arquivo no Drive com a nova dimensão arquivo.setContent(imagemRedimensionada.getBytes()); arquivo.setName("OK_" + arquivo.getName()); // Marca como processada } } } } /** * Remove todas as linhas totalmente vazias do intervalo com dados * para manter a planilha limpa e organizada. */ function removerLinhasVaziasAppSheet() { var planilha = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); }); // Se a linha estiver completamente vazia, deleta a linha if (linhaVazia) { planilha.deleteRow(i + 1); } } }
