function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Cadastro_Filhos"); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Aba Cadastro_Filhos não encontrada"})) .setMimeType(ContentService.MimeType.JSON); } var rows = sheet.getDataRange().getValues(); var headers = rows[0]; // Identifica dinamicamente onde está a coluna do ID (ID_Filho ou id) var colId = -1; for (var h = 0; h < headers.length; h++) { var headerName = String(headers[h]).trim().toLowerCase(); if (headerName === 'id_filho' || headerName === 'id') { colId = h; break; } } // Se não achar pelo cabeçalho, assume que o ID está na primeira coluna (índice 0) if (colId === -1) colId = 0; var acao = data.acao; // 'cadastrar', 'editar', 'excluir' var idEnviado = String(data.id || "").trim(); if (acao === 'excluir') { var encontrado = false; // Percorre de baixo para cima para evitar problemas de deslocamento de índices ao deletar for (var i = rows.length - 1; i >= 1; i--) { var currentId = String(rows[i][colId]).trim(); if (currentId === idEnviado) { sheet.deleteRow(i + 1); // Deleta a linha inteira fisicamente da planilha encontrado = true; } } if (encontrado) { return ContentService.createTextOutput(JSON.stringify({"status": "success", "message": "Linha excluída com sucesso"})) .setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "ID não encontrado para exclusão"})) .setMimeType(ContentService.MimeType.JSON); } } // Dados que serão salvos na ordem padrão das colunas var rowData = [ data.id || ('ID_' + new Date().getTime()), data.nome_completo || "", data.data_nascimento || "", data.telefone || "", data.email || "", data.endereco || "", data.data_entrada || "", data.cargo_funcao || "", data.orixa_frente || "", data.status_ativo || "Ativo", data.perfil || "Membro", data.observacoes || "" ]; if (acao === 'editar') { var atualizado = false; for (var i = 1; i < rows.length; i++) { if (String(rows[i][colId]).trim() === idEnviado) { sheet.getRange(i + 1, 1, 1, rowData.length).setValues([rowData]); atualizado = true; break; } } if (!atualizado) { sheet.appendRow(rowData); } } else { // Cadastrar novo sheet.appendRow(rowData); } return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
