/** * ============================================================================ * GOOGLE APPS SCRIPT - TENDA DE ZÉ CIGANO * Arquivo: Codigo.gs * ============================================================================ */ // Configuração do Nome da Aba de Membros const NOME_ABA_MEMBROS = "Cadastro_Filhos"; /** * FUNÇÃO: doGet * Descrição: Recebe requisições GET para teste ou retorno rápido de dados. */ function doGet(e) { return ContentService.createTextOutput(JSON.stringify({ status: "sucesso", mensagem: "Web App ativa!" })) .setMimeType(ContentService.MimeType.JSON); } /** * FUNÇÃO: doPost * Descrição: Ponto de entrada para operações de Inserção, Edição e Exclusão. */ function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(NOME_ABA_MEMBROS); if (!sheet) { sheet = ss.insertSheet(NOME_ABA_MEMBROS); // Criar cabeçalhos padrão caso a aba não exista sheet.appendRow([ "ID_Filho", "Nome_Completo", "Data_Nascimento", "Telefone", "Email", "Endereco", "Data_Entrada", "Cargo_Funcao", "Orixa_Frente", "Status_Ativo", "Perfil", "Observacoes" ]); } const payload = JSON.parse(e.postData.contents); // ROTA DE EXCLUSÃO if (payload.acao === 'excluir' || payload.action === 'delete') { return excluirMembro(sheet, payload.id); } // ROTA DE SALVAR OU ATUALIZAR return salvarOuAtualizarMembro(sheet, payload); } catch (erro) { return ContentService.createTextOutput(JSON.stringify({ status: 'erro', mensagem: 'Falha no processamento: ' + erro.toString() })).setMimeType(ContentService.MimeType.JSON); } } /** * FUNÇÃO DESTAQUE: salvarOuAtualizarMembro * Descrição: Procura se o ID do membro já existe. Se existir, faz UPDATE da linha. * Se não existir, gera um novo ID e faz INSERT (appendRow). */ function salvarOuAtualizarMembro(sheet, dados) { const data = sheet.getDataRange().getValues(); const idProcurado = dados.id ? String(dados.id).trim() : ""; let linhaEncontrada = -1; if (idProcurado) { for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === idProcurado) { linhaEncontrada = i + 1; // Linha real na planilha break; } } } // Define ID novo ou mantém o existente const idMembro = idProcurado ? idProcurado : "MB-" + new Date().getTime(); const linhaDados = [ idMembro, dados.nome_completo || "", dados.data_nascimento || "", dados.telefone || "", dados.email || "", dados.endereco || "", dados.data_entrada || "", dados.cargo_funcao || "", dados.orixa_frente || "", dados.status_ativo || "Ativo", dados.perfil || "Membro", dados.observacoes || "" ]; if (linhaEncontrada > 0) { // ATUALIZAÇÃO (UPDATE) sheet.getRange(linhaEncontrada, 1, 1, linhaDados.length).setValues([linhaDados]); return ContentService.createTextOutput(JSON.stringify({ status: 'sucesso', acao: 'atualizacao', id: idMembro, mensagem: 'Membro atualizado com sucesso!' })).setMimeType(ContentService.MimeType.JSON); } else { // INSERÇÃO (INSERT) sheet.appendRow(linhaDados); return ContentService.createTextOutput(JSON.stringify({ status: 'sucesso', acao: 'insercao', id: idMembro, mensagem: 'Novo membro cadastrado com sucesso!' })).setMimeType(ContentService.MimeType.JSON); } } /** * FUNÇÃO DESTAQUE: excluirMembro * Descrição: Localiza e remove fisicamente a linha do membro da planilha. */ function excluirMembro(sheet, id) { const data = sheet.getDataRange().getValues(); const idProcurado = String(id).trim(); let linhaEncontrada = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === idProcurado) { linhaEncontrada = i + 1; break; } } if (linhaEncontrada > 0) { sheet.deleteRow(linhaEncontrada); return ContentService.createTextOutput(JSON.stringify({ status: 'sucesso', mensagem: 'Membro removido da planilha com sucesso!' })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ status: 'erro', mensagem: 'ID não localizado para exclusão.' })).setMimeType(ContentService.MimeType.JSON); } }
