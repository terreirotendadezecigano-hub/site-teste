/** * ============================================================================ * GOOGLE APPS SCRIPT - TENDA DE ZÉ CIGANO * Arquivo: codigo.gs - Versão com Varredura Recursiva Direta na Pasta Central * ============================================================================ */ const ID_PASTA_CENTRAL = "1QcgIX_ZpWyfOL_PHQ5JsJ-2Ga7IePqIL"; function doGet(e) { try { var sheetName = e.parameter.sheet; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!sheetName) { return ContentService.createTextOutput(JSON.stringify({ status: "erro", mensagem: "Parâmetro 'sheet' não informado." })) .setMimeType(ContentService.MimeType.JSON); } var sheet = ss.getSheetByName(sheetName); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ status: "erro", mensagem: "Aba não encontrada: " + sheetName })) .setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); if (data.length <= 1) { return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); } var headers = data[0]; var rows = data.slice(1); var result = rows.map(function(row) { var obj = {}; headers.forEach(function(header, index) { var val = row[index]; var headerUpper = header.toString().trim().toUpperCase(); if (typeof val === "string" && val.trim() !== "") { var valTrim = val.trim(); if (valTrim.includes("drive.google.com") || valTrim.includes("docs.google.com") || valTrim.includes("googleusercontent.com")) { var match = valTrim.match(/\/d\/([a-zA-Z0-9_-]+)/) || valTrim.match(/id=([a-zA-Z0-9_-]+)/); if (match && match[1]) { if (headerUpper.endsWith("_URL") && (headerUpper.includes("AUDIO") || headerUpper.includes("VIDEO") || headerUpper.includes("PONTO"))) { val = "https://drive.google.com/uc?export=download&id=" + match[1]; } else { val = "https://lh3.googleusercontent.com/d/" + match[1]; } } } } obj[header] = val; }); return obj; }); return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "erro", mensagem: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function processarTodasTabelasURL() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var abas = ss.getSheets(); abas.forEach(function(aba) { var dados = aba.getDataRange().getValues(); if (dados.length <= 1) return; var cabecalho = dados[0]; var colunasURLIndices = []; for (var colIdx = 0; colIdx < cabecalho.length; colIdx++) { var nomeColuna = cabecalho[colIdx].toString().trim().toUpperCase(); if (nomeColuna.endsWith("_URL")) { colunasURLIndices.push({ index: colIdx, nome: nomeColuna }); } } if (colunasURLIndices.length === 0) return; for (var i = 1; i < dados.length; i++) { colunasURLIndices.forEach(function(colInfo) { var urlFinal = ""; var nomeArquivoBusca = ""; for (var c = 0; c < cabecalho.length; c++) { var hClean = cabecalho[c].toString().trim().toUpperCase(); var valCell = dados[i][c]; if (valCell && valCell.toString().trim() !== "") { var valStr = valCell.toString().trim(); if (valStr.indexOf("http://") === 0 || valStr.indexOf("https://") === 0) { if (hClean.includes("LINK") || hClean.includes("VIDEO") || hClean.includes("AUDIO") || hClean.includes("ARQUIVO") || hClean.includes("PONTO")) { urlFinal = valStr; } } else if (valStr.includes(".") || valStr.includes("/")) { nomeArquivoBusca = valStr.split("/").pop().trim(); } } } if (urlFinal && (urlFinal.includes("drive.google.com") || urlFinal.includes("googleusercontent.com"))) { var match = urlFinal.match(/\/d\/([a-zA-Z0-9_-]+)/) || urlFinal.match(/id=([a-zA-Z0-9_-]+)/); if (match && match[1]) { if (colInfo.nome.includes("AUDIO") || colInfo.nome.includes("VIDEO") || colInfo.nome.includes("PONTO")) { urlFinal = "https://drive.google.com/uc?export=download&id=" + match[1]; } else { urlFinal = "https://lh3.googleusercontent.com/d/" + match[1]; } } } else if (nomeArquivoBusca) { var arquivo = buscarArquivoRecursivo_(ID_PASTA_CENTRAL, nomeArquivoBusca); if (arquivo) { try { arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e) {} var fileId = arquivo.getId(); if (colInfo.nome.includes("AUDIO") || colInfo.nome.includes("VIDEO") || colInfo.nome.includes("PONTO")) { urlFinal = "https://drive.google.com/uc?export=download&id=" + fileId; } else { urlFinal = "https://lh3.googleusercontent.com/d/" + fileId; } } } if (urlFinal) { aba.getRange(i + 1, colInfo.index + 1).setValue(urlFinal); } }); } }); Logger.log("Processamento concluído com sucesso!"); } function buscarArquivoRecursivo_(idPasta, nomeArquivo) { try { var pasta = DriveApp.getFolderById(idPasta); var arquivos = pasta.getFilesByName(nomeArquivo); if (arquivos.hasNext()) { return arquivos.next(); } var subpastas = pasta.getFolders(); while (subpastas.hasNext()) { var sub = subpastas.next(); var encontrado = buscarArquivoRecursivo_(sub.getId(), nomeArquivo); if (encontrado) return encontrado; } } catch (e) {} return null; } function doPost(e) { try { var payload = JSON.parse(e.postData.contents); var nomeAba = payload.nomeAba; var valores = payload.valores; var ss = SpreadsheetApp.getActiveSpreadsheet(); var aba = ss.getSheetByName(nomeAba); if (!aba) { return ContentService.createTextOutput(JSON.stringify({ "status": "erro", "mensagem": "Aba não encontrada." })).setMimeType(ContentService.MimeType.JSON); } aba.appendRow(valores); processarTodasTabelasURL(); return ContentService.createTextOutput(JSON.stringify({ "status": "sucesso", "mensagem": "Gravado!" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "status": "erro", "mensagem": error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
