/** Lista eventos públicos futuros; normaliza datas que o Sheets entrega como Date. */
function TZC_M04_listarAgendaPublica() {
  try {
    const cache=CacheService.getScriptCache(), saved=cache.get('TZC_M04_AGENDA_PUBLICA_V02');
    if(saved)return JSON.parse(saved);
    const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID);
    const aba=ss.getSheetByName(TZC_M04_CFG_V01.ABA_EVENTOS);
    if(!aba||aba.getLastRow()<2)return {ok:true,eventos:[]};
    const rows=tzcM04_lerObjetosV01_(aba);
    const hoje=Utilities.formatDate(new Date(),TZC_M04_CFG_V01.FUSO,'yyyy-MM-dd');
    const eventos=rows.map(e=>({origem:e,dataISO:tzcM04_dataISO_V02_(e.Data_Inicio)}))
      .filter(x=>x.dataISO&&x.dataISO.slice(0,10)>=hoje&&
        tzcM04_simV01_(x.origem.Status)&&tzcM04_simV01_(x.origem.Acesso_Publico)&&
        x.origem.ID_Evento&&x.origem.Titulo)
      .sort((a,b)=>a.dataISO.localeCompare(b.dataISO))
      .slice(0,30).map(x=>({id:String(x.origem.ID_Evento),titulo:String(x.origem.Titulo),
        tipo:String(x.origem.Tipo||''),data:x.dataISO,local:String(x.origem.Local||''),
        descricao:String(x.origem.Descricao||''),limite:tzcM04_dataISO_V02_(x.origem.Data_Limite_Confirmacao),
        lanche:tzcM04_simV01_(x.origem.Oferece_Lanche)}));
    const result={ok:true,eventos:eventos};
    cache.put('TZC_M04_AGENDA_PUBLICA_V02',JSON.stringify(result),30);
    return result;
  } catch(e) {
    console.error('M04 listarAgendaPublica: '+e.message);
    return {ok:false,mensagem:'Não foi possível carregar a agenda agora.',eventos:[]};
  }
}

/** Registra intenção de membro logado ou previsão de visitante em aba separada. */
function TZC_M04_responderEvento(json) {
  const p = tzcM04_jsonV01_(json);
  if (!p) return {ok:false,mensagem:'Resposta inválida.'};
  const eventoId = tzcM04_textoV01_(p.eventoId,100);
  const resposta = tzcM04_respostaV01_(p.resposta);
  const qtd = Math.max(0,Math.min(TZC_M04_CFG_V01.MAX_PESSOAS,Math.floor(Number(p.quantidade)||0)));
  if (!eventoId || !resposta || (resposta!=='Não vou' && qtd<1)) return {ok:false,mensagem:'Escolha uma resposta e informe quantas pessoas irão.'};
  const evento = tzcM04_eventoV01_(eventoId);
  if (!evento || !tzcM04_simV01_(evento.Status)) return {ok:false,mensagem:'Este evento não está aberto para respostas.'};
  if (tzcM04_fechadoV01_(evento) || tzcM04_iniciadoV01_(evento)) return {ok:false,mensagem:'O prazo de resposta para este evento terminou.'};

  let tipo='Visitante', user=null, nome='', email='', whatsapp='';
  if (String(p.tipo||'').toLowerCase()==='membro') {
    const sessao = tzcM02_sessaoV03_(p.token);
    if (!sessao) return {ok:false,mensagem:'Entre na área de membros para registrar sua resposta.'};
    user = tzcM02_buscarUsuarioV03_(sessao.email);
    if (!user || user.status!=='ativo') return {ok:false,mensagem:'Seu acesso não está ativo.'};
    tipo='Membro'; nome=user.nome; email=user.email;
  } else {
    if (!tzcM04_simV01_(evento.Acesso_Publico)) return {ok:false,mensagem:'Este evento está disponível somente para membros.'};
    nome=tzcM04_textoV01_(p.nome,100);
    email=tzcM04_emailV01_(p.email);
    whatsapp=tzcM04_whatsappV04_(p.whatsapp);
    if (!nome || !email || !whatsapp) return {ok:false,mensagem:'Informe nome, e-mail e WhatsApp com DDD.'};
    if (p.aceitaContatoEvento!==true) return {ok:false,mensagem:'Confirme que podemos usar esses contatos para avisos deste evento.'};
  }

  try {
    const lock=LockService.getScriptLock(); lock.waitLock(8000);
    try {
      const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID);
      const sh=ss.getSheetByName(TZC_M04_CFG_V01.ABA_RESPOSTAS);
      if (!sh || sh.getLastRow()<1) throw new Error('Execute TZC_M04_prepararEstrutura antes do uso.');
      const map=tzcM04_indicesV01_(sh), rows=sh.getLastRow()>1?sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getValues():[];
      const key=tipo==='Membro'?String(user.id):email.toLowerCase(); let rowNo=0;
      for (let i=0;i<rows.length;i++) {
        const r=rows[i];
        if (String(r[map.ID_Evento]||'')===eventoId && String(r[map.Tipo_Participante]||'')===tipo &&
          (tipo==='Membro'?String(r[map.ID_Usuario]||'')===key:(String(r[map.Email_Visitante]||'').toLowerCase()===key&&(!String(r[map.WhatsApp_Visitante]||'')||tzcM04_whatsappV04_(r[map.WhatsApp_Visitante])===whatsapp)))) {rowNo=i+2;break;}
      }
      const now=new Date(), id=rowNo?sh.getRange(rowNo,map.ID_Resposta+1).getValue():'RSP_'+Utilities.getUuid();
      const values={ID_Resposta:id,ID_Evento:eventoId,Tipo_Participante:tipo,ID_Usuario:user?user.id:'',
        Nome_Visitante:tipo==='Visitante'?nome:'',Email_Visitante:tipo==='Visitante'?email:'',WhatsApp_Visitante:tipo==='Visitante'?whatsapp:'',
        Aceita_Contato_Evento:tipo==='Visitante'?'Sim':'Não',Resposta:resposta,
        Qtd_Participantes:resposta==='Não vou'?0:qtd,Observacoes:tzcM04_textoV01_(p.observacoes,500),
        Data_Resposta:rowNo?(rows[rowNo-2][map.Data_Resposta]||now):now,Atualizado_Em:now,
        Status:'Ativa',Lembrete_Presenca_Enviado_Em:rowNo?(rows[rowNo-2][map.Lembrete_Presenca_Enviado_Em]||''):'',
        Lembrete_Email_Visitante_Aceito:tipo==='Visitante'&&p.aceitaLembrete?'Sim':'Não',
        Lembrete_Evento_Enviado_Em:rowNo?(rows[rowNo-2][map.Lembrete_Evento_Enviado_Em]||''):''};
      tzcM04_gravarObjetoV01_(sh,map,rowNo||sh.getLastRow()+1,values);
      return {ok:true,mensagem:'Resposta salva. Obrigado!',tipo:tipo,quantidade:values.Qtd_Participantes};
    } finally {lock.releaseLock();}
  } catch(e) { console.error('M04 responderEvento: '+e.message); return {ok:false,mensagem:'Não foi possível salvar sua resposta agora.'}; }
}

/** Retorna, em uma única consulta, somente os dados pessoais da sessão ativa. */
function TZC_M04_obterMinhaArea(token) {
  const user=tzcM04_usuarioSessaoV01_(token);
  if(!user)return {ok:false,mensagem:'Entre novamente para abrir sua área.'};
  try {
    const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID);
    const eventsSh=ss.getSheetByName(TZC_M04_CFG_V01.ABA_EVENTOS);
    const responsesSh=ss.getSheetByName(TZC_M04_CFG_V01.ABA_RESPOSTAS);
    const presenceSh=ss.getSheetByName(TZC_M04_CFG_V01.ABA_PRESENCAS);
    const duesSh=ss.getSheetByName(TZC_M04_CFG_V01.ABA_MENSALIDADES);
    const id=String(user.id), now=Utilities.formatDate(new Date(),TZC_M04_CFG_V01.FUSO,'yyyy-MM-dd');
    const events=eventsSh?tzcM04_lerObjetosV01_(eventsSh):[];
    const myResponses=Object.create(null);
    if(responsesSh&&responsesSh.getLastRow()>1){
      const m=tzcM04_indicesV01_(responsesSh), rows=responsesSh.getRange(2,1,responsesSh.getLastRow()-1,responsesSh.getLastColumn()).getDisplayValues();
      rows.forEach(r=>{if(String(r[m.ID_Usuario]||'')===id&&String(r[m.Status]||'Ativa').toLowerCase()!=='cancelada')myResponses[String(r[m.ID_Evento])]={resposta:String(r[m.Resposta]||''),quantidade:Number(r[m.Qtd_Participantes]||0),atualizadoEm:String(r[m.Atualizado_Em]||'')};});
    }
    const upcoming=events.filter(e=>tzcM04_simV01_(e.Status)&&e.ID_Evento&&e.Titulo&&tzcM04_dataISO_V02_(e.Data_Inicio).slice(0,10)>=now&&!tzcM04_iniciadoV01_(e))
      .sort((a,b)=>tzcM04_dataISO_V02_(a.Data_Inicio).localeCompare(tzcM04_dataISO_V02_(b.Data_Inicio))).slice(0,12)
      .map(e=>({id:String(e.ID_Evento),titulo:String(e.Titulo),data:tzcM04_dataISO_V02_(e.Data_Inicio),local:String(e.Local||''),resposta:myResponses[String(e.ID_Evento)]||null,ofereceLanche:tzcM04_simV01_(e.Oferece_Lanche)}));
    const eventsById=Object.create(null);events.forEach(e=>eventsById[String(e.ID_Evento)]=e);
    const presencas=[];
    if(presenceSh&&presenceSh.getLastRow()>1){
      const m=tzcM04_indicesV01_(presenceSh), rows=presenceSh.getRange(2,1,presenceSh.getLastRow()-1,presenceSh.getLastColumn()).getDisplayValues();
      rows.forEach(r=>{if(String(r[m.ID_Usuario]||'')!==id)return;const ev=eventsById[String(r[m.ID_Evento]||'')];presencas.push({evento:ev?String(ev.Titulo):'Evento',data:ev?tzcM04_dataISO_V02_(ev.Data_Inicio):'',resposta:String(r[m.Resposta]||''),situacao:String(r[m.Presenca]||r[m.Status]||'Pendente'),confirmadoEm:String(r[m.Confirmado_Em]||''),checkinEm:String(r[m.Checkin_Em]||'')});});
    }
    presencas.sort((a,b)=>(b.data||'').localeCompare(a.data||''));
    const mensalidades=[];
    if(duesSh&&duesSh.getLastRow()>1){
      const m=tzcM04_indicesV01_(duesSh), rows=duesSh.getRange(2,1,duesSh.getLastRow()-1,duesSh.getLastColumn()).getDisplayValues();
      rows.forEach(r=>{if(String(r[m.ID_Usuario]||'')!==id)return;mensalidades.push({competencia:String(r[m.Competencia]||''),vencimento:String(r[m.Data_Vencimento]||''),valor:String(r[m.Valor]||''),status:String(r[m.Status]||'Em aberto'),dataPagamento:String(r[m.Data_Pagamento]||''),formaPagamento:String(r[m.Forma_Pagamento]||'')});});
    }
    mensalidades.sort((a,b)=>(b.vencimento||'').localeCompare(a.vencimento||''));
    return {ok:true,usuario:{id:id,nome:String(user.nome||''),perfil:String(user.perfil||'')},eventos:upcoming,presencas:presencas,mensalidades:mensalidades};
  } catch(e){console.error('M04 obterMinhaArea: '+e.message);return {ok:false,mensagem:'Não foi possível carregar sua área agora.'};}
}

/** Após o evento, o filho registra uma declaração; ela fica pendente de validação. */
function TZC_M04_registrarCheckin(token, eventoId, relato) {
  const sessao=tzcM02_sessaoV03_(token), evento=tzcM04_eventoV01_(eventoId);
  if (!sessao) return {ok:false,mensagem:'Entre na área de membros para continuar.'};
  const user=tzcM02_buscarUsuarioV03_(sessao.email);
  if (!user || user.status!=='ativo') return {ok:false,mensagem:'Seu acesso não está ativo.'};
  if (!evento || !tzcM04_encerradoV01_(evento)) return {ok:false,mensagem:'O check-in fica disponível depois do término do evento.'};
  const declaracao=String(relato||'').trim()==='Não compareci'?'Não compareci':'Compareci';
  try {
    const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID), sh=ss.getSheetByName(TZC_M04_CFG_V01.ABA_PRESENCAS);
    if (!sh) throw new Error('A aba Presencas não foi encontrada.');
    const map=tzcM04_indicesV01_(sh), rows=sh.getLastRow()>1?sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getValues():[];
    for(let i=0;i<rows.length;i++) if(String(rows[i][map.ID_Evento]||'')===String(eventoId)&&String(rows[i][map.ID_Usuario]||'')===String(user.id)) {
      if(String(rows[i][map.Presenca]||'').toLowerCase()==='presente'||String(rows[i][map.Presenca]||'').toLowerCase()==='ausente')
        return {ok:true,mensagem:'Sua presença já foi validada pela gestão.'};
      const row=i+2, now=new Date();
      tzcM04_gravarObjetoV01_(sh,map,row,{Presenca:'Pendente',Status:'Pendente de validação',Resposta:declaracao,
        Checkin_Em:now,Atualizado_Em:now,Confirmado_Por:'',Confirmado_Em:''});
      return {ok:true,mensagem:'Aviso recebido. A presença ficou pendente para validação.'};
    }
    const row=sh.getLastRow()+1, now=new Date();
    tzcM04_gravarObjetoV01_(sh,map,row,{ID_Presenca:'PRE_'+Utilities.getUuid(),ID_Evento:String(eventoId),ID_Usuario:String(user.id),
      Data_Resposta:now,Resposta:declaracao,Presenca:'Pendente',Observacoes:'',Criado_Em:now,Atualizado_Em:now,
      Nome_Visitante:'',WhatsApp_Visitante:'',Email_Visitante:'',Tipo_Participante:'Membro',Aceita_Lembrete:'Sim',
      Contato_Lembrete:user.email,Status:'Pendente de validação',Confirmado_Por:'',Checkin_Em:now,Confirmado_Em:''});
    return {ok:true,mensagem:'Aviso recebido. A presença ficou pendente para validação.'};
  } catch(e) { console.error('M04 registrarCheckin: '+e.message); return {ok:false,mensagem:'Não foi possível registrar seu aviso agora.'}; }
}

/** Lista pendências somente para dirigente/administrador ou delegado daquele evento. */
function TZC_M04_listarPendencias(token, eventoId) {
  const caller=tzcM04_usuarioSessaoV01_(token); if(!caller) return {ok:false,mensagem:'Entre novamente.'};
  if(!tzcM04_podeGerirEventoV01_(caller,eventoId)) return {ok:false,mensagem:'Você não está autorizado para este evento.'};
  try {
    const sh=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID).getSheetByName(TZC_M04_CFG_V01.ABA_PRESENCAS);
    if(!sh||sh.getLastRow()<2) return {ok:true,registros:[]};
    const map=tzcM04_indicesV01_(sh), rows=sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getDisplayValues();
    const users=tzcM04_mapaNomesV01_();
    const records=rows.map((r,i)=>({linha:i+2,id:r[map.ID_Presenca],evento:r[map.ID_Evento],idUsuario:r[map.ID_Usuario],
      nome:users[String(r[map.ID_Usuario]||'')]||'Membro',relato:r[map.Resposta],presenca:r[map.Presenca],
      checkin:r[map.Checkin_Em],status:r[map.Status]})).filter(r=>r.evento===String(eventoId)&&
        String(r.checkin||'')&&['pendente','presente','ausente'].includes(String(r.presenca||'').toLowerCase()));
    return {ok:true,registros:records};
  } catch(e) {console.error('M04 listarPendencias: '+e.message);return {ok:false,mensagem:'Não foi possível carregar a lista.'};}
}

/** Retorna eventos encerrados com check-in pendente que o responsável pode revisar. */
function TZC_M04_listarEventosGestao(token) {
  const caller=tzcM04_usuarioSessaoV01_(token); if(!caller)return {ok:false,mensagem:'Entre novamente.',eventos:[]};
  try {
    const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID), ev=ss.getSheetByName(TZC_M04_CFG_V01.ABA_EVENTOS), pr=ss.getSheetByName(TZC_M04_CFG_V01.ABA_PRESENCAS);
    if(!ev||!pr||pr.getLastRow()<2)return {ok:true,eventos:[]};
    const events=tzcM04_lerObjetosV01_(ev), pm=tzcM04_indicesV01_(pr), rows=pr.getRange(2,1,pr.getLastRow()-1,pr.getLastColumn()).getDisplayValues();
    const pending=Object.create(null); rows.forEach(r=>{if(String(r[pm.Checkin_Em]||'')&&r[pm.ID_Evento])pending[String(r[pm.ID_Evento])]=true;});
    const out=events.filter(e=>pending[String(e.ID_Evento)]&&tzcM04_encerradoV01_(e)&&tzcM04_podeGerirEventoV01_(caller,e.ID_Evento))
      .map(e=>({id:String(e.ID_Evento),titulo:String(e.Titulo),data:String(e.Data_Inicio||''),local:String(e.Local||'')}));
    return {ok:true,eventos:out};

}

/** Gera prévia e PDF A4 de previsão, sem expor contatos pessoais dos participantes. */

/** Confirma/corrige a presença e audita o responsável que alterou o resultado. */
function TZC_M04_confirmarCheckin(token, presencaId, resultado) {

    const ss=SpreadsheetApp.openById(TZC_M04_CFG_V01.PLANILHA_ID), sh=ss.getSheetByName(TZC_M04_CFG_V01.ABA_PRESENCAS);
    if(!sh||sh.getLastRow()<2)return {ok:false,mensagem:'Registro não encontrado.'};
    const map=tzcM04_indicesV01_(sh), data=sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).getValues();
    let row=0, old=null;
    for(let i=0;i<data.length;i++)if(String(data[i][map.ID_Presenca]||'')===String(presencaId)){row=i+2;old=data[i];break;}

/**
 * Converte datas do Google Sheets em texto ISO local, para que a agenda e o
 * botão de resposta encontrem o mesmo evento mesmo quando a célula é Date.
 */
function tzcM04_dataISO_V02_(value) {
  if(value===null||value===undefined||value==='')return '';
  let d;
  if(value instanceof Date)d=value;
  else {
    const s=String(value).trim();
    const match=s.match(/^(\d{4})-(\d{2})-(\d{2})(?:[ T](\d{1,2}):(\d{2})(?::(\d{2}))?)?/);
    if(match)return match[1]+'-'+match[2]+'-'+match[3]+(match[4]?' '+('0'+match[4]).slice(-2)+':'+match[5]+(match[6]?':'+match[6]:''):'');
    d=new Date(s);
  }
  if(!(d instanceof Date)||isNaN(d.getTime()))return '';
  return Utilities.formatDate(d,TZC_M04_CFG_V01.FUSO,'yyyy-MM-dd HH:mm:ss');
}
function tzcM04_dataV01_(raw) {
  const s=String(raw||'').trim(); if(!s)return null;
  const m=s.match(/^(\d{4})-(\d{2})-(\d{2})(?:\s+(\d{1,2}):(\d{2}))?/);
  // Converte explicitamente o horário de Brasília (-03) para UTC, independente
  // do fuso configurado no projeto Apps Script ou na planilha.
  if(m)return new Date(Date.UTC(Number(m[1]),Number(m[2])-1,Number(m[3]),Number(m[4]||0)+3,Number(m[5]||0)));
  const d=new Date(s); return isNaN(d.getTime())?null:d;
}
function tzcM04_lerObjetosV01_(sh) {

  const headers=vals[0].map(v=>String(v||'').trim());
