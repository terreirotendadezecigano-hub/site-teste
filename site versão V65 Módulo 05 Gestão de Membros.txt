  /* A home mostra uma única Mensagem do Dia: a publicação cuja data está mais próxima de hoje.
     Outras publicações públicas continuam aparecendo em seguida, sem duplicar mensagens diárias. */
  /* Escolhe uma única mensagem diária e mantém os outros conteúdos sem duplicá-la. */
  function separarPublicacoesHome(lista){
    var registros=Array.isArray(lista)?lista.slice():[];
    var mensagens=registros.filter(function(p){return /mensagem\s+do\s+dia/i.test(String(p.tipo||''))});
    function dataMs(p){
      var v=String(p.data||'').trim(),m=v.match(/^(\d{4})-(\d{2})-(\d{2})/);
      if(m)return new Date(Number(m[1]),Number(m[2])-1,Number(m[3])).getTime();
      m=v.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
      if(m)return new Date(Number(m[3]),Number(m[2])-1,Number(m[1])).getTime();
      return Number.POSITIVE_INFINITY;
    }
    var hoje=new Date();hoje.setHours(0,0,0,0);
    mensagens.sort(function(a,b){
      var da=dataMs(a),db=dataMs(b),dif=Math.abs(da-hoje.getTime())-Math.abs(db-hoje.getTime());
      return dif||db-da;
    });
    var selecionada=mensagens[0]||null;
    var outras=registros.filter(function(p){return !/mensagem\s+do\s+dia/i.test(String(p.tipo||''))}).slice(0,2);
    return {mensagem:selecionada,outras:outras};
  }
